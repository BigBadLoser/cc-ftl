import { BodyC } from '../body';
import type { Sim } from '../sim';
import { Struct } from '../structure';
import { DIRS, STANDOFF } from './geometry';
import { MiningPodC } from './components';

const HEADINGS: readonly (readonly number[])[] = [[0, 2], [4, 6], [1, 3, 5, 7]];

// Where a hauler waits to unload at `target`: same standoff ring, but the nearest bay that no other hauler holds.
export function dockSpot(sim: Sim, hauler: number, target: { x: number; y: number }, size: number): { x: number; y: number } {
  const others: { x: number; y: number }[] = [];
  for (const id of sim.world.query(MiningPodC)) {
    const s = id !== hauler ? sim.world.get(id, MiningPodC)?.bay : null;
    if (s) others.push(s);
  }
  const cx = target.x + 0.5, cy = target.y + 0.5;
  const taken = (x: number, y: number) => others.some((o) => Math.abs(o.x - x) + Math.abs(o.y - y) < size);
  const g = sim.sector.grid;
  const roofed = (x: number, y: number) => {
    const tx = Math.floor(x), ty = Math.floor(y);
    if (tx < 0 || ty < 0 || tx >= sim.sector.width || ty >= sim.sector.height) return true;
    return g.get('roof', tx, ty) !== 0 || g.get('struct', tx, ty) === Struct.Wall;
  };
  const at = sim.world.get(hauler, BodyC) ?? { x: cx, y: cy };
  const bays = HEADINGS.map((group) =>
    group
      .map((k, i) => ({ x: cx + (DIRS[k] as readonly [number, number])[0] * STANDOFF, y: cy + (DIRS[k] as readonly [number, number])[1] * STANDOFF, i }))
      .sort((a, b) => Math.abs(a.x - at.x) + Math.abs(a.y - at.y) - (Math.abs(b.x - at.x) + Math.abs(b.y - at.y)) || a.i - b.i),
  ).flat();
  const hit = bays.find((c) => !roofed(c.x, c.y) && !taken(c.x, c.y)) ?? bays.find((c) => !taken(c.x, c.y)) ?? bays[0];
  return hit ? { x: hit.x, y: hit.y } : { x: cx, y: cy };
}
