type Rect = { x0: number; y0: number; x1: number; y1: number };
type Size = { w: number; h: number };
const hit = (a: Rect, b: Rect): boolean => a.x0 < b.x1 && b.x0 < a.x1 && a.y0 < b.y1 && b.y0 < a.y1;

// Where the "you are here" arrow of the minimap sits: around a ship, nearest side that stays clear of the other labels.
export function placeArrow(ship: Rect, size: Size, view: Size, labels: Rect[], gap = 6, margin = 4): { x: number; y: number } | null {
  if (ship.x1 <= 0 || ship.y1 <= 0 || ship.x0 >= view.w || ship.y0 >= view.h) return null;
  const cx = (ship.x0 + ship.x1) / 2 - size.w / 2;
  const cy = (ship.y0 + ship.y1) / 2 - size.h / 2;
  const sides = [
    { x: cx, y: ship.y0 - gap - size.h },
    { x: cx, y: ship.y1 + gap },
    { x: ship.x0 - gap - size.w, y: cy },
    { x: ship.x1 + gap, y: cy },
  ];
  const clamp = (v: number, size2: number, max: number) => Math.max(margin, Math.min(v, max - size2 - margin));
  let best: { x: number; y: number } | null = null;
  let bestCost = Infinity;
  for (const s of sides) {
    const x = clamp(s.x, size.w, view.w);
    const y = clamp(s.y, size.h, view.h);
    const r = { x0: x, y0: y, x1: x + size.w, y1: y + size.h };
    const cost = (hit(r, ship) ? 500 : 0) + labels.filter((o) => hit(r, o)).length * 10 + Math.hypot(x - s.x, y - s.y) * 0.5;
    if (cost < bestCost) {
      best = { x, y };
      bestCost = cost;
    }
  }
  return best;
}
