type Spot = { x: number; y: number };

// Give every crew member their own spot, nearest first.
export function allocateSpots(crew: readonly { id: number; x: number; y: number }[], spots: readonly Spot[]): Map<number, Spot> {
  const result = new Map<number, Spot>();
  if (spots.length === 0) return result;
  const pending = [...crew].sort((p, q) => p.id - q.id);
  for (let pass = 0; pending.length > 0; pass++) {
    const open = spots.map((s, n) => ({ s, n }));
    const group = pending.splice(0, open.length);
    const scored: { dist: number; person: (typeof group)[number]; n: number }[] = [];
    for (const person of group) for (const o of open) scored.push({ dist: Math.abs(person.x - o.s.x) + Math.abs(person.y - o.s.y), person, n: o.n });
    scored.sort((p, q) => p.dist - q.dist || p.person.id - q.person.id || p.n - q.n);
    const taken = new Set<number>();
    for (const c of scored) {
      if (result.has(c.person.id) || taken.has(c.n)) continue;
      result.set(c.person.id, spots[c.n] as Spot);
      taken.add(c.n);
    }
    if (pass > 50) break;
  }
  return result;
}
