import { BodyC } from '../body';
import type { Sim } from '../sim';
import { Struct } from '../structure';
import { DIRS, STANDOFF } from './geometry';
import { MiningPodC } from './components';

const HEADINGS: readonly (readonly number[])[] = [[0, 2], [4, 6], [1, 3, 5, 7]];

export function hoverPoint(sim: Sim, craft: number, goal: { x: number; y: number }, radius: number): { x: number; y: number } {
  const taken: { x: number; y: number }[] = [];
  for (const id of sim.world.query(MiningPodC)) {
    const p = id !== craft ? sim.world.get(id, MiningPodC)?.spot : null;
    if (p) taken.push(p);
  }
  const mx = goal.x + 0.5, my = goal.y + 0.5;
  const vacant = (x: number, y: number) => taken.every((t) => (t.x - x) ** 2 + (t.y - y) ** 2 >= radius * radius - 1e-9);
  const field = sim.sector.grid;
  const passable = (x: number, y: number) => {
    const col = Math.floor(x), row = Math.floor(y);
    if (col < 0 || row < 0 || col >= sim.sector.width || row >= sim.sector.height) return false;
    const s = field.get('struct', col, row);
    return s === Struct.Space || s === Struct.Ground;
  };
  const here = sim.world.get(craft, BodyC) ?? { x: mx, y: my };
  const options = HEADINGS.map((set) =>
    set
      .map((h, n) => ({ x: mx - (DIRS[h] as readonly [number, number])[0] * STANDOFF, y: my - (DIRS[h] as readonly [number, number])[1] * STANDOFF, n }))
      .sort((p, q) => (p.x - here.x) ** 2 + (p.y - here.y) ** 2 - ((q.x - here.x) ** 2 + (q.y - here.y) ** 2) || p.n - q.n),
  ).flat();
  const pick = options.find((o) => passable(o.x, o.y) && vacant(o.x, o.y)) ?? options.find((o) => vacant(o.x, o.y)) ?? options[0];
  return pick ? { x: pick.x, y: pick.y } : { x: mx, y: my };
}
