import type { Sim } from '../sim';
import { TICK_HZ } from '../time';
import { BOND_FADE_SEC, setScore } from './bonds';

const towardZero = (score: number): number => (score > 0 ? Math.max(0, score - 1) : Math.min(0, score + 1));

function cool(world: Sim, from: number, to: number, entry: { score: number; touched: number; kind?: string }, period: number): boolean {
  const steps = Math.floor((world.tick - entry.touched) / period);
  if (steps <= 0) return false;
  for (let i = 0; i < steps; i++) {
    entry.touched += period;
    setScore(world, from, to, entry as never, towardZero(entry.score));
  }
  return true;
}

export function cooldownFeelings(world: Sim): void {
  const period = TICK_HZ * BOND_FADE_SEC;
  Object.keys(world.bonds.opinions).forEach((f) => {
    const row = world.bonds.opinions[+f]!;
    Object.keys(row).forEach((t) => {
      const entry = row[+t]!;
      const touched = cool(world, +f, +t, entry, period);
      if (touched && !entry.score && !entry.kind) delete row[+t];
    });
    if (!Object.keys(row).length) delete world.bonds.opinions[+f];
  });
}
